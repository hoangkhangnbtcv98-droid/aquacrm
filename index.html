2. Mã Google Apps Script (Code.gs)
  1. Mở file Google Sheets dataCRM của bạn.
  2. Chọn menu Tiện ích mở rộng (Extensions) → Apps Script.
  3. Xóa mã cũ, dán toàn bộ đoạn mã bên dưới vào và bấm Lưu (Ctrl+S).
  4. Bấm Triển khai (Deploy) → Tùy chọn triển khai mới (New deployment) → Chọn loại Ứng dụng web (Web app) → Quyền truy cập: Bất kỳ ai (Anyone) → Copy URL dán lên ô số 1.
// Mã Google Apps Script đồng bộ 100% với file dataCRM
const SHEETS = ['DON_HANG', 'DON_HANG_CT', 'DS_DL', 'DS_KH', 'SAN_PHAM', 'SAN_PHAM_DT', 'Du_Lieu'];

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const result = {};
  SHEETS.forEach(name => {
    const sheet = ss.getSheetByName(name);
    if (sheet) {
      const data = sheet.getDataRange().getValues();
      if (data.length > 1) {
        const headers = data[0].map(h => String(h).trim());
        result[name] = data.slice(1).filter(r => r.join('') !== '').map(row => {
          const obj = {};
          headers.forEach((h, i) => {
            if (h) obj[h] = row[i];
          });
          return obj;
        });
      } else {
        result[name] = [];
      }
    }
  });
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: result }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const action = payload.action; // 'append' hoặc 'appendBatch'
    
    if (action === 'append') {
      appendRowToSheet(ss, payload.sheetName, payload.rowData);
    } else if (action === 'appendOrder') {
      // Ghi đồng thời DON_HANG và nhiều dòng DON_HANG_CT
      appendRowToSheet(ss, 'DON_HANG', payload.order);
      if (Array.isArray(payload.items)) {
        payload.items.forEach(item => appendRowToSheet(ss, 'DON_HANG_CT', item));
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function appendRowToSheet(ss, sheetName, rowObj) {
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return;
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  const newRow = headers.map(h => rowObj[h] !== undefined ? rowObj[h] : '');
  sheet.appendRow(newRow);
}
